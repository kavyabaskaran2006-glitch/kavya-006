//Prime or not
import java.util.Scanner;
class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        System.out.print("Enter value :");
        int a=sc.nextInt();
        int i=2,f=1;
        while(i<a){
            if(a%i==0){
                f=0;
                break;
            }
            i++;
        }
        System.out.println(f==1?"Prime":"not prime");
    }
}
