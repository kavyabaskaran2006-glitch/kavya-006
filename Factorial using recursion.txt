// Factorial using recursion
import java.util.Scanner;
class Main{
    public int fact(int n){
        if (n==0 || n==1){
            return 1;
        }
        else{
            return n*fact(n-1);
        }
    }
    public static void main(String[] args){
       Scanner scan=new Scanner(System.in);
       System.out.print("Enter a value:");
       int a=scan.nextInt();
       Main obj=new Main();
       System.out.println(obj.fact(a));
    }
}
