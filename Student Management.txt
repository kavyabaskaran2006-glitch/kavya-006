
//Student Management

class Student {
    String name;
    int rollNo;
    float marks;

    void display() {
        System.out.println("Name: " + name);
        System.out.println("Roll No: " + rollNo);
        System.out.println("Marks: " + marks);
    }
    public static void main(String[] args) {
        Student s1 = new Student();
        s1.name = "Kavya";
        s1.rollNo = 101;
        s1.marks = 89.5f;

        Student s2 = new Student();
        s2.name = "Sri";
        s2.rollNo = 102;
        s2.marks = 91.0f;

        s1.display();
        s2.display();
    }
}
